      MQ([RocketMQ 消息到达<br/>eventId · sellerId · productId])
      --> SEL[按 event_id 查审计表]

      --> UPD1[置 callback_status = SUCCESS]
      --> ACK1([ack])
      ST -- "已终态<br/>（已上架/已驳回/已转人工）" --> ACK2([直接 ack])

      %% ============ 主分支：记录不存在 ============
      HIT -- "否（新事件）" --> CHK2[Feign 查 catalog 商品状态<br/>状态守卫：防生产者重复事件]
      --> ST2{"商品 ON_REVIEW?"}
      ST2 -- "否" --> ACK3([直接 ack])
      ST2 -- "是" --> GRAPH["跑审核图 · 6 节点<br/>LLM 调用统一走 LlmInvoker：<br/>错误分类 → 瞬时错误重试 → 耗尽降级转人工"]
      --> INS[(INSERT 审计记录<br/>uk_event_id 唯一索引<br/>callback_status = PENDING)]
      --> DUP{"唯一键冲突?"}
      DUP -- "冲突（并发同消息）" --> ROUTE[转重入分支处理<br/>查已有记录]
      DUP -- "插入成功" --> CB[Feign 回调 catalog<br/>改商品状态]
      --> CBR{"回调结果?"}
      CBR -- "成功" --> UPD2[置 SUCCESS]
      --> DONE([处理完成 · ack])
      CBR -- "失败" --> UPD3[置 FAILED]
      --> THROW[抛异常]
      --> MQ2([MQ 重投<br/>回到起点 · 命中重入分支<br/>只重放回调不重跑图])

      %% ============ 定时补偿 ============
      subgraph COMP["定时补偿 · 每分钟"]
          SCAN["扫描条件：<br/>callback_status IN PENDING, FAILED<br/>retry_count < 5<br/>updated_at 早于 5 分钟"]
          --> LOOP{"重放回调"}
          LOOP -- "成功" --> S1[置 SUCCESS]
          LOOP -- "失败" --> RC[retry_count + 1]
          --> C5{">= 5 次?"}
          C5 -- "否" --> KEEP[保持 FAILED<br/>下轮继续扫描]
          C5 -- "是" --> AB[置 ABANDONED<br/>error 日志 · 人工介入]
      end

      UPD3 -. 补偿兜底 .-> SCAN
