package org.monkey.agent.review.service.component.aiservice;

import cn.hutool.dfa.WordTree;
import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

/**
 * 文本规则仓库：加载违禁品词库 / 极限词库 / 排除上下文 / 敏感信息正则
 * 词库文件在 resources/text-rules/ 下，每行一条，空行和 # 开头的注释行忽略
 */
@Component
@Slf4j
public class TextRuleRepository {

    // 违禁品词文件：强证据词，命中无条件驳回
    private static final String BANNED_PATH = "/text-rules/banned-words.txt";
    // 极限词文件：广告法极限词，命中驳回（排除上下文内跳过）
    private static final String LIMIT_PATH = "/text-rules/limit-words.txt";
    // 排除上下文文件：极限词的合法搭配（如"第一章"），匹配前先挖掉
    private static final String EXCLUSION_PATH = "/text-rules/limit-exclusions.txt";
    // 敏感信息正则文件：每行一条正则（手机号/微信/QQ/URL）
    private static final String SENSITIVE_PATH = "/text-rules/sensitive-patterns.txt";

    // 违禁品词列表
    private List<String> bannedWords = List.of();
    // 极限词列表
    private List<String> limitWords = List.of();
    // 排除上下文列表
    private List<String> limitExclusions = List.of();
    // 敏感信息正则列表（编译好的 Pattern）
    private List<Pattern> sensitivePatterns = List.of();
    // 违禁品 DFA 匹配树
    private WordTree bannedTree;
    // 极限词 DFA 匹配树
    private WordTree limitTree;

    /**
     * 启动加载全部词库：文件缺失抛异常（fail-fast），内容为空允许（词库运营起步期）
     */
    @PostConstruct
    public void init() {
        bannedWords = loadLines(BANNED_PATH);
        limitWords = loadLines(LIMIT_PATH);
        limitExclusions = loadLines(EXCLUSION_PATH);
        sensitivePatterns = loadPatterns(SENSITIVE_PATH);
        // WordTree：DFA 多模匹配树——一段文本一次遍历命中词库所有词，复杂度 O(文本长度)
        bannedTree = new WordTree();
        bannedTree.addWords(bannedWords);
        limitTree = new WordTree();
        limitTree.addWords(limitWords);
        log.info("[TextRuleRepository] 词库加载完成: 违禁品={}, 极限词={}, 排除={}, 正则={}",
                bannedWords.size(), limitWords.size(), limitExclusions.size(), sensitivePatterns.size());
    }

    /**
     * 读词库文件：每行一个词，去掉首尾空白，忽略空行和 # 注释行
     */
    private List<String> loadLines(String path) {
        List<String> lines = new ArrayList<>();
        InputStream in = getClass().getResourceAsStream(path);
        if (in == null) {
            throw new IllegalStateException("词库文件不存在: " + path);
        }
        BufferedReader reader = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8));
        reader.lines().forEach(line -> {
            String word = line.trim();
            if (!word.isEmpty() && !word.startsWith("#")) {
                lines.add(word);
            }
        });
        return lines;
    }

    /**
     * 读正则文件：每行一条正则，编译失败抛异常（fail-fast，坏正则不该静默跳过）
     */
    private List<Pattern> loadPatterns(String path) {
        List<Pattern> patterns = new ArrayList<>();
        for (String line : loadLines(path)) {
            patterns.add(Pattern.compile(line));
        }
        return patterns;
    }

    // ========== 访问器 ==========

    public List<String> bannedWords() {
        return bannedWords;
    }

    public List<String> limitWords() {
        return limitWords;
    }

    public List<String> limitExclusions() {
        return limitExclusions;
    }

    public List<Pattern> sensitivePatterns() {
        return sensitivePatterns;
    }

    public WordTree bannedTree() {
        return bannedTree;
    }

    public WordTree limitTree() {
        return limitTree;
    }
}

package org.monkey.agent.review.service.component.aiservice;

import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.regex.Pattern;

/**
 * 文本审核第一层：规则筛查（词库 + 正则，零 LLM 成本）
 * 处理顺序：违禁品词 → 极限词（先挖排除上下文）→ 敏感信息，任一命中立即短路
 */
@Component
@Slf4j
public class TextRuleScreener {

    // 规则层判定：REJECT=命中硬规则驳回 / UNDECIDED=干净进入 LLM 层
    public enum Verdict { REJECT, UNDECIDED }

    // 命中详情：词库来源 + 命中词 + 驳回文案（人工复核台按 source 分流）
    @Data
    @AllArgsConstructor
    public static class Hit {
        private String source;   // 词库来源：banned 违禁品 / limit 极限词 / sensitive 敏感信息
        private String word;     // 命中的词或正则原文
        private String detail;   // 驳回理由原文
    }

    @Autowired
    private TextRuleRepository repository;

    /**
     * 规则筛查：返回 Hit 表示命中驳回，null 表示干净（进入 LLM 层）
     */
    public Hit screen(String title, String description) {
        String text = title + " " + description;

        // 1. 违禁品词：无条件驳回
        List<String> bannedHits = matchAll(repository.bannedTree(), text);
        if (!bannedHits.isEmpty()) {
            return new Hit("banned", bannedHits.get(0), "含违禁品词: " + bannedHits);
        }

        // 2. 极限词：先挖掉排除上下文（"第一章"里的"第一"是合法用法）
        String cleaned = text;
        for (String exclusion : repository.limitExclusions()) {
            cleaned = cleaned.replace(exclusion, " ");
        }
        List<String> limitHits = matchAll(repository.limitTree(), cleaned);
        if (!limitHits.isEmpty()) {
            return new Hit("limit", limitHits.get(0), "含广告法极限词: " + limitHits);
        }

        // 3. 敏感信息引流：正则
        for (Pattern pattern : repository.sensitivePatterns()) {
            if (pattern.matcher(text).find()) {
                return new Hit("sensitive", pattern.pattern(), "含引流信息: " + pattern.pattern());
            }
        }

        return null;
    }

    /**
     * DFA 匹配（加锁）：WordTree match 内部使用共享状态，并发调用同步——审核并发低，开销可忽略
     */
    private synchronized List<String> matchAll(cn.hutool.dfa.WordTree tree, String text) {
        return tree.matchAll(text);
    }

    /**
     * 违禁词参考文本：LLM 找变体时的对照词库（空格 join）
     */
    public String bannedWordsRef() {
        return String.join(" ", repository.bannedWords());
    }
}

package org.monkey.agent.review.service.component.node;

import cn.hutool.core.collection.CollUtil;
import lombok.extern.slf4j.Slf4j;
import org.bsc.langgraph4j.action.NodeAction;
import org.monkey.agent.review.service.component.aiservice.TextJudgeService;
import org.monkey.agent.review.service.component.aiservice.TextRuleScreener;
import org.monkey.agent.review.service.component.enums.Verdict;
import org.monkey.agent.review.service.component.model.ReviewState;
import org.monkey.agent.review.service.component.result.TextJudgeResult;
import org.monkey.agent.review.service.component.result.TextReviewResult;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Map;

/**
 * 文本审核节点：三层方案
 * 第一层规则层（词库+正则，零 LLM 成本）→ 第二层 LLM 变体识别 → 第三层置信度兜底
 */
@Component
@Slf4j
public class TextReviewNode implements NodeAction<ReviewState> {

    // 兜底阈值（与类目链一致）：PASS 方向严（错放行平台担责）、REJECT 方向宽（错杀商家重提可挽回）
    static final double PASS_CONFIDENCE = 0.9;
    static final double REJECT_CONFIDENCE = 0.7;

    @Autowired
    private TextRuleScreener textRuleScreener;

    @Autowired
    private TextJudgeService textJudgeService;

    @Override
    public Map<String, Object> apply(ReviewState state) throws Exception {
        log.info("[Node][TextReviewNode] apply");
        // 读：走访问器
        String title = state.title();
        String description = state.description();

        // ========== 第一层：规则层（词库 + 正则，零 LLM 成本） ==========
        TextRuleScreener.Hit hit = textRuleScreener.screen(title, description);
        if (hit != null) {
            log.info("[Node][TextReviewNode] rule reject: {}", hit);
            return Map.of(ReviewState.TEXT_RESULT, buildRejectResult(hit));
        }

        // ========== 第二层：LLM 变体识别 ==========
        TextJudgeResult judge = textJudgeService.judge(title, description, textRuleScreener.bannedWordsRef());
        log.info("[Node][TextReviewNode] judge: {}", judge);

        // ========== 第三层：置信度兜底（裁决规则表） ==========
        // 归一化：verdict 白名单校验，非法值按 ESCALATED 处理
        Verdict verdict = normalizeVerdict(judge.getVerdict());
        // 置信度防御：NaN/越界按 0 处理，自动落到转人工分支
        double confidence = validConfidence(judge.getConfidence());

        // PASS 方向：严格兜底（≥0.9 才通过）
        if (verdict == Verdict.APPROVED && confidence >= PASS_CONFIDENCE) {
            return Map.of(ReviewState.TEXT_RESULT,
                    new TextReviewResult(Verdict.APPROVED, List.of(), judge.getRiskPhrases(), judge.getReason()));
        }
        // REJECT 方向：宽松兜底（≥0.7）+ 必须有变体证据（riskPhrases 非空）
        if (verdict == Verdict.REJECTED && CollUtil.isNotEmpty(judge.getRiskPhrases()) && confidence >= REJECT_CONFIDENCE) {
            // 变体词并入 bannedWordsHit：规则层和 LLM 层的命中证据合并
            return Map.of(ReviewState.TEXT_RESULT,
                    new TextReviewResult(Verdict.REJECTED, judge.getRiskPhrases(), judge.getRiskPhrases(), judge.getReason()));
        }
        // 其余全部转人工：PASS 低置信 / REJECT 证据不足 / ESCALATE / 输出异常
        return Map.of(ReviewState.TEXT_RESULT,
                new TextReviewResult(Verdict.ESCALATED, List.of(), judge.getRiskPhrases(), buildEscalateReason(verdict, judge)));
    }

    /**
     * 规则层驳回结果：bannedWordsHit 放命中词，detail 放驳回文案
     */
    private TextReviewResult buildRejectResult(TextRuleScreener.Hit hit) {
        return new TextReviewResult(Verdict.REJECTED, List.of(hit.getWord()), List.of(), hit.getDetail());
    }

    /**
     * verdict 归一化：LLM 输出白名单校验，非法值按 ESCALATED 处理（包可见 static 便于单测）
     */
    static Verdict normalizeVerdict(String rawVerdict) {
        if ("PASS".equalsIgnoreCase(rawVerdict)) {
            return Verdict.APPROVED;
        }
        if ("REJECT".equalsIgnoreCase(rawVerdict)) {
            return Verdict.REJECTED;
        }
        if ("ESCALATE".equalsIgnoreCase(rawVerdict)) {
            return Verdict.ESCALATED;
        }
        // 非法值：LLM 输出不可信，宁可转人工
        log.warn("[Node][TextReviewNode] LLM 输出非法 verdict: {}", rawVerdict);
        return Verdict.ESCALATED;
    }

    /**
     * 置信度防御：NaN/越界按 0 处理，自动落到转人工分支
     */
    static double validConfidence(double rawConfidence) {
        if (Double.isNaN(rawConfidence) || rawConfidence < 0 || rawConfidence > 1) {
            return 0;
        }
        return rawConfidence;
    }

    /**
     * 转人工理由分流：人工复核台按 reason 前缀分类处理
     */
    private String buildEscalateReason(Verdict verdict, TextJudgeResult judge) {
        if (verdict == Verdict.APPROVED) {
            // 精判说通过但置信度不足
            return "精判置信度不足 " + judge.getConfidence() + "：" + judge.getReason();
        }
        if (verdict == Verdict.REJECTED) {
            // 驳回但说不出变体证据
            return "驳回证据不足：riskPhrases 缺失，LLM 理由：" + judge.getReason();
        }
        // ESCALATED：LLM 自己说无法判断（含输出非法值场景）
        return "LLM 无法判断或输出异常：" + judge.getReason();
    }
}
